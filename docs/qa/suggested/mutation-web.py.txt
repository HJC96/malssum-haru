def run(m):
    mid,desc,f,old,new=m
    d=f"{SP}/wmut/{mid}"
    shutil.rmtree(d,ignore_errors=True)
    os.makedirs(d)
    subprocess.run(["rsync","-a","--exclude","node_modules","--exclude","src/qa","--exclude","dist",SRC+"/",d+"/"],check=True)
    os.symlink("/Users/hanjichan/Desktop/git/malssum-haru/node_modules",d+"/../node_modules") if not os.path.exists(d+"/../node_modules") else None
    os.symlink("/Users/hanjichan/Desktop/git/malssum-haru/web/node_modules",d+"/node_modules")
    p=os.path.join(d,f)
    s=open(p,encoding="utf-8").read()
    if old not in s: return (mid,desc,"NOT-APPLIED","")
    open(p,"w",encoding="utf-8").write(s.replace(old,new,1))
    r=subprocess.run(["npx","vitest","run"],cwd=d,capture_output=True,text=True,timeout=300)
    out=r.stdout+r.stderr
    names=sorted(set(re.findall(r"FAIL\s+(\S+\.test\.tsx?)",out)))
    n=re.findall(r"Tests\s+(.*)",out)
    return (mid,desc,"KILLED" if r.returncode!=0 else "SURVIVED",(", ".join(names)+" "+(n[0] if n else ""))[:250])
