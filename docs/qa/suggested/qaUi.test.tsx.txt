import { render, screen, within } from '@testing-library/react';
import userEvent from '@testing-library/user-event';
import { describe, expect, it } from 'vitest';
import { App } from '@/app/App';
import { QtProviderCard } from '@/components/QtProviderCard';
import { normalizeQtToday } from '@/app/qt/normalize';
import { QT_MOCK_SCENARIOS, maeilConfirmed, durannoConfirmed } from '@/app/qt/fixtures';
import { I18nProvider } from '@/i18n';
import { NOW } from '@/app/test/render';

const card = (raw: unknown, now: Date, lang: 'ko' | 'en' = 'ko') => {
  const p = normalizeQtToday({ schemaVersion: '1', generatedAt: '', providers: [raw] }).providers[0]!;
  return render(<I18nProvider initialLang={lang}><QtProviderCard provider={p} now={now} /></I18nProvider>);
};

describe('QA UI', () => {
  it('OBSERVE stale tab after Seoul midnight: confirmed 09-24 card rendered at 09-25 01:30 KST', () => {
    const { container } = card(maeilConfirmed(), new Date('2026-09-24T16:30:00Z'));
    console.log('STALE-TAB TEXT:', container.textContent);
    console.log('TZ', Intl.DateTimeFormat().resolvedOptions().timeZone);
  });
  it('OBSERVE DATE_MISMATCH with yesterday providerDate', () => {
    const raw = { ...maeilConfirmed(), providerDate: '2026-09-23', availabilityStatus: 'RANGE_UNAVAILABLE', reasonCode: 'DATE_MISMATCH', passage: null, displayReference: null, verifiedAt: null, notice: null };
    const { container } = card(raw, NOW);
    console.log('MISMATCH TEXT:', container.textContent);
  });
  it('OBSERVE RANGE_NOT_PERMITTED date-specific link (no date shown)', () => {
    const raw = { ...durannoConfirmed(), providerDate: null, availabilityStatus: 'RANGE_NOT_PERMITTED', reasonCode: 'PERMISSION_UNCONFIRMED', passage: null, displayReference: null, verifiedAt: null, notice: null };
    const { container } = card(raw, NOW, 'en');
    console.log('NOTPERMITTED EN TEXT:', container.textContent);
  });
  it('EN confirmed shows Korean-source notes', () => {
    const { container } = card(maeilConfirmed(), NOW, 'en');
    console.log('EN CONFIRMED:', container.textContent);
    expect(container.textContent).toMatch(/Korean/);
  });
  it('all links: https + target/rel', async () => {
    render(<App initialLang="en" qtFetcher={async () => normalizeQtToday(QT_MOCK_SCENARIOS.confirmed())} now={NOW} />);
    await screen.findAllByRole('article');
    for (const a of Array.from(document.querySelectorAll('a[href]'))) {
      const href = a.getAttribute('href')!;
      if (href.startsWith('#')) continue;
      expect(href.startsWith('https://')).toBe(true);
      expect(a.getAttribute('rel')).toContain('noopener');
      expect(a.getAttribute('target')).toBe('_blank');
    }
  });
  it('language switch keeps plan name & dates (form state)', async () => {
    render(<App initialLang="ko" qtFetcher={async () => normalizeQtToday(QT_MOCK_SCENARIOS.confirmed())} now={NOW} />);
    await screen.findAllByRole('article');
    const inputs = Array.from(document.querySelectorAll('input'));
    console.log('INPUT COUNT', inputs.length, inputs.map((i) => `${i.type}:${i.name || i.id}:${i.autocomplete}`).slice(0, 12).join(' | '));
    const before = inputs.map((i) => (i as HTMLInputElement).value);
    await userEvent.click(screen.getByRole('button', { name: 'English' }));
    const after = Array.from(document.querySelectorAll('input')).map((i) => (i as HTMLInputElement).value);
    expect(after).toEqual(before);
  });
  it('storage/cookie/url untouched after typing plan name and reading input', async () => {
    render(<App initialLang="ko" qtFetcher={async () => normalizeQtToday(QT_MOCK_SCENARIOS.confirmed())} now={NOW} />);
    await screen.findAllByRole('article');
    const urlBefore = location.href;
    const text = Array.from(document.querySelectorAll('input[type=text]'))[0] as HTMLInputElement;
    await userEvent.type(text, 'PRIVATE-NAME-XYZ');
    expect(localStorage.length).toBe(0);
    expect(sessionStorage.length).toBe(0);
    expect(document.cookie).toBe('');
    expect(location.href).toBe(urlBefore);
    expect(JSON.stringify(Object.entries(localStorage))).not.toContain('PRIVATE');
  });
});
