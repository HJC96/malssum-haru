def run(m):
    mid,desc,f,old,new=m
    d=f"{SP}/mut3/{mid}"
    shutil.rmtree(d,ignore_errors=True)
    shutil.copytree(SRC,d,ignore=shutil.ignore_patterns("target"))
    p=os.path.join(d,f)
    s=open(p,encoding="utf-8").read()
    if old not in s: return (mid,desc,"NOT-APPLIED","")
    open(p,"w",encoding="utf-8").write(s.replace(old,new,1))
    argline="-Dhttps.proxyHost=127.0.0.1 -Dhttps.proxyPort=9 -Dhttp.proxyHost=127.0.0.1 -Dhttp.proxyPort=9"
    r=subprocess.run(["mvn","-o","-q","-B","test",f"-DargLine={argline}"],cwd=d,capture_output=True,text=True,timeout=600)
    out=r.stdout+r.stderr
    failed=sorted(set(re.findall(r"^\[ERROR\]\s+(?:Failures|Errors):?.*$|^\[ERROR\]\s+(\S+Test\.\S+)",out,re.M)))
    names=sorted(set(re.findall(r"\[ERROR\]\s+(\w+Test\.\w+)",out)))
    comp="COMPILE" if "COMPILATION ERROR" in out else ""
    return (mid,desc,("KILLED" if r.returncode!=0 else "SURVIVED")+(" "+comp if comp else ""),", ".join(names)[:300])
